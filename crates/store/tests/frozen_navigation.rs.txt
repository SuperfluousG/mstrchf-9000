fn visit_HardRequirement(v: &HardRequirement) {
    match v {
        HardRequirement::AllergenExclusion { code } => visit_AllergenCode(code),
        HardRequirement::Diet { code } => visit_DietCode(code),
        HardRequirement::IngredientExclusion { ingredient_id } => {
            let _: &str = ingredient_id.value();
        }
    }
}
fn visit_RestrictionDeclaration(v: &RestrictionDeclaration) {
    match v {
        RestrictionDeclaration::Unconfirmed {} => (),
        RestrictionDeclaration::ConfirmedNone { confirmed_at } => {
            let _: &str = confirmed_at.value();
        }
        RestrictionDeclaration::Confirmed {
            confirmed_at,
            requirements,
        } => {
            let _: &str = confirmed_at.value();
            for requirement in requirements {
                visit_HardRequirement(requirement);
            }
        }
    }
}
macro_rules! diners {
    ($diners:expr) => {
        for diner in $diners {
            visit_AgeBand(diner.age_band());
            visit_RestrictionDeclaration(diner.restrictions());
            visit_PortionPresetName(diner.portions().preset());
            for meal in diner.portions().meal_overrides() {
                visit_MealType(meal.meal_type());
                visit_PortionPresetName(meal.preset());
            }
        }
    };
}
macro_rules! preferences {
    ($p:expr) => {{
        let p = $p;
        for e in p.equipment() {
            visit_EquipmentCode(e);
        }
        for meal in p.meal_times() {
            visit_MealType(meal.meal_type());
        }
        visit_MobilityTemperature(p.mobility().temperature());
        visit_SpendingStyle(p.spending_style());
    }};
}
macro_rules! connections {
    ($c:expr) => {{
        let c = $c;
        for delivery in c.delivery_preferences() {
            visit_Weekday(delivery.weekday());
        }
        if let Some(provider) = c.harness() {
            visit_Provider(provider);
        }
        visit_LoyaltyStatus(c.loyalty());
        for channel in c.reminders().channels() {
            visit_ReminderChannel(channel);
        }
        for prompt in c.reminders().prompts() {
            visit_ReminderPrompt(prompt);
        }
        if let Some(retailer) = c.retailer() {
            visit_Retailer(retailer);
        }
    }};
}
macro_rules! usual_week {
    ($w:expr) => {{
        let w = $w;
        for cell in w.cells() {
            visit_MealLocation(cell.choice());
            visit_MealType(cell.meal_type());
            visit_Weekday(cell.weekday());
        }
        for meal in w.enabled_meal_types() {
            visit_MealType(meal);
        }
    }};
}
pub fn inspect(s: &mstrchf_store::__SNAPSHOT__) {
    let _: &[Never] = s.weeks();
    let _: &[Never] = s.conflicts();
    for v in s.weeks().iter().chain(s.conflicts()) {
        visit_never(v);
    }
    for job in s.jobs() {
        visit_Action(job.action());
        visit_JobState(job.state());
        if let Some(reason) = job.reason() {
            visit_ErrorCode(reason);
        }
        if let Some(action) = job.next_action() {
            visit_RequiredAction(action);
        }
    }
    let draft = s.household_draft();
    for stage in draft.completed_stages() {
        visit_HouseholdStage(stage);
    }
    diners!(draft.diners());
    if let Some(p) = draft.preferences() {
        preferences!(p);
    }
    if let Some(c) = draft.connections() {
        connections!(c);
    }
    if let Some(w) = draft.usual_week() {
        usual_week!(w);
    }
    if let Some(profile) = s.profile() {
        diners!(profile.diners());
        preferences!(profile.preferences());
        connections!(profile.connections());
        usual_week!(profile.usual_week());
    }
}
